120template <auto... Members>
121constexpr auto uniqueMemberNames() -> bool
122{
123 const auto nameStorage = std::tuple{reflectedMemberName<Members>()...};
124 const auto names =
125 std::apply([](const auto&... name) { return std::array<std::string_view, sizeof...(Members)>{name.view()...}; },
126 nameStorage);
127 for (std::size_t left = 0; left < names.size(); ++left)
128 {
129 for (std::size_t right = left + 1; right < names.size(); ++right)
130 {
131 if (names[left].compare(names[right]) == 0)
132 {
133 return false;
134 }
135 }
136 }
137 return true;
138}

503 /**
504 * @brief Resolves a C++ field type to its logical database type at compile
505 * time. Users may extend the closed set through LogicalTypeTraits<T>.
506 */
507 template <typename T>
508 consteval auto logicalType() -> ColumnType
509 {
510 using field_t = std::remove_cvref_t<T>;
511 using value_t = std::remove_cv_t<detail::static_optional_value_t<field_t>>;
512
513 if constexpr (requires { LogicalTypeTraits<value_t>::value; })
514 {

517 else

519 static_assert(detail::alwaysFalse<value_t>,
520 "ORM_MODEL_LOGICAL_TYPE: unsupported model field type; specialize "
521 "orm::model::LogicalTypeTraits<T>.");
