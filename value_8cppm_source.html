102 [[nodiscard]] static auto fromStorage(model::ColumnType logicalType, Value value) -> QueryValue
103 {
104 if (not isCompatibleStorage(logicalType, value))

107 }

109 return QueryValue{logicalType, std::move(value), StorageValueTag{}};

112 /**
113 * @brief Checks both the canonical variant alternative and its logical range.
114 */
115 [[nodiscard]] static auto isCompatibleStorage(model::ColumnType logicalType,
116 const Value& storedValue) noexcept -> bool
117 {
118 const auto* intValue = std::get_if<int>(&storedValue);
119 const auto* unsignedValue = std::get_if<unsigned long long>(&storedValue);
120 const auto* doubleValue = std::get_if<double>(&storedValue);
121
122 switch (logicalType)
123 {
124 case model::ColumnType::Bool:
125 return intValue != nullptr and (*intValue == 0 or *intValue == 1);
126 case model::ColumnType::Char:
127 return intValue != nullptr and *intValue >= std::numeric_limits<signed char>::lowest() and
128 *intValue <= std::numeric_limits<unsigned char>::max();
129 case model::ColumnType::UnsignedChar:
130 return intValue != nullptr and *intValue >= 0 and
131 *intValue <= std::numeric_limits<unsigned char>::max();
132 case model::ColumnType::Short:
133 return intValue != nullptr and *intValue >= std::numeric_limits<short>::lowest() and
134 *intValue <= std::numeric_limits<short>::max();
135 case model::ColumnType::UnsignedShort:
136 return intValue != nullptr and *intValue >= 0 and
137 *intValue <= std::numeric_limits<unsigned short>::max();
138 case model::ColumnType::Int:
139 return intValue != nullptr;
140 case model::ColumnType::UnsignedInt:
141 return unsignedValue != nullptr and *unsignedValue <= std::numeric_limits<unsigned int>::max();
142 case model::ColumnType::LongLong:
143 return std::holds_alternative<long long>(storedValue);
144 case model::ColumnType::UnsignedLongLong:
145 return unsignedValue != nullptr;
146 case model::ColumnType::Float:
147 return doubleValue != nullptr and std::isfinite(*doubleValue) and
148 *doubleValue >= static_cast<double>(std::numeric_limits<float>::lowest()) and
149 *doubleValue <= static_cast<double>(std::numeric_limits<float>::max()) and
150 static_cast<double>(static_cast<float>(*doubleValue)) == *doubleValue;
151 case model::ColumnType::Double:
152 return doubleValue != nullptr and std::isfinite(*doubleValue);
153 case model::ColumnType::String:
154 return std::holds_alternative<std::string>(storedValue);
155 case model::ColumnType::Date:
156 return compatibleValue<Date>(storedValue);
157 case model::ColumnType::TimeOfDay:
158 return compatibleValue<TimeOfDay>(storedValue);
159 case model::ColumnType::LocalDateTime:
160 return compatibleValue<LocalDateTime>(storedValue);
161 case model::ColumnType::UtcTimestamp:
162 return compatibleValue<UtcTimestamp>(storedValue);
163 case model::ColumnType::Uuid:
164 return compatibleValue<Uuid>(storedValue);
165 }
166
167 return false;
168 }
