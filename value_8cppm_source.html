107 [[nodiscard]] static auto isCompatibleStorage(model::ColumnType logicalType,
108 const Value& storedValue) noexcept -> bool
109 {
110 const auto* intValue = std::get_if<int>(&storedValue);
111 const auto* unsignedValue = std::get_if<unsigned long long>(&storedValue);
112 const auto* doubleValue = std::get_if<double>(&storedValue);
113
114 switch (logicalType)
115 {
116 case model::ColumnType::Bool:
117 return intValue != nullptr and (*intValue == 0 or *intValue == 1);
118 case model::ColumnType::Char:
119 return intValue != nullptr and *intValue >= std::numeric_limits<signed char>::lowest() and
120 *intValue <= std::numeric_limits<unsigned char>::max();
121 case model::ColumnType::UnsignedChar:
122 return intValue != nullptr and *intValue >= 0 and
123 *intValue <= std::numeric_limits<unsigned char>::max();
124 case model::ColumnType::Short:
125 return intValue != nullptr and *intValue >= std::numeric_limits<short>::lowest() and
126 *intValue <= std::numeric_limits<short>::max();
127 case model::ColumnType::UnsignedShort:
128 return intValue != nullptr and *intValue >= 0 and
129 *intValue <= std::numeric_limits<unsigned short>::max();
130 case model::ColumnType::Int:
131 return intValue != nullptr;
132 case model::ColumnType::UnsignedInt:
133 return unsignedValue != nullptr and *unsignedValue <= std::numeric_limits<unsigned int>::max();
134 case model::ColumnType::LongLong:
135 return std::holds_alternative<long long>(storedValue);
136 case model::ColumnType::UnsignedLongLong:
137 return unsignedValue != nullptr;
138 case model::ColumnType::Float:
139 return doubleValue != nullptr and std::isfinite(*doubleValue) and
140 *doubleValue >= static_cast<double>(std::numeric_limits<float>::lowest()) and
141 *doubleValue <= static_cast<double>(std::numeric_limits<float>::max()) and
142 static_cast<double>(static_cast<float>(*doubleValue)) == *doubleValue;
143 case model::ColumnType::Double:
144 return doubleValue != nullptr and std::isfinite(*doubleValue);
145 case model::ColumnType::String:
146 return std::holds_alternative<std::string>(storedValue);
147 case model::ColumnType::Uuid:
148 return false;
149 }
150
151 return false;
152 }
