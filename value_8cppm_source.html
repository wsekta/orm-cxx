110 /**
111 * @brief Restores a canonical transport value with its model-level type.
112 *
113 * Hydration and backend adapters use this factory when several logical
114 * column types share the same SOCI exchange representation.
115 */
116 [[nodiscard]] static auto fromStorage(model::ColumnType logicalType, Value value) -> QueryValue
117 {
118 if (not isCompatibleStorage(logicalType, value))
119 {
120 throw std::invalid_argument{"Query value storage does not match its logical column type"};
121 }
122
123 return QueryValue{logicalType, std::move(value), StorageValueTag{}};
124 }
125
126 /**
127 * @brief Checks both the canonical variant alternative and its logical range.
128 */
129 [[nodiscard]] static auto isCompatibleStorage(model::ColumnType logicalType,
130 const Value& storedValue) noexcept -> bool
131 {
132 const auto* intValue = std::get_if<int>(&storedValue);
133 const auto* unsignedValue = std::get_if<unsigned long long>(&storedValue);
134 const auto* doubleValue = std::get_if<double>(&storedValue);
135
136 switch (logicalType)
137 {
138 case model::ColumnType::Bool:
139 return intValue != nullptr and (*intValue == 0 or *intValue == 1);
140 case model::ColumnType::Char:
141 return intValue != nullptr and *intValue >= std::numeric_limits<signed char>::lowest() and
142 *intValue <= std::numeric_limits<unsigned char>::max();
143 case model::ColumnType::UnsignedChar:
144 return intValue != nullptr and *intValue >= 0 and
145 *intValue <= std::numeric_limits<unsigned char>::max();
146 case model::ColumnType::Short:
147 return intValue != nullptr and *intValue >= std::numeric_limits<short>::lowest() and
148 *intValue <= std::numeric_limits<short>::max();
149 case model::ColumnType::UnsignedShort:
150 return intValue != nullptr and *intValue >= 0 and
151 *intValue <= std::numeric_limits<unsigned short>::max();
152 case model::ColumnType::Int:
153 return intValue != nullptr;
154 case model::ColumnType::UnsignedInt:
155 return unsignedValue != nullptr and *unsignedValue <= std::numeric_limits<unsigned int>::max();
156 case model::ColumnType::LongLong:
157 return std::holds_alternative<long long>(storedValue);
158 case model::ColumnType::UnsignedLongLong:
159 return unsignedValue != nullptr;
160 case model::ColumnType::Float:
161 return doubleValue != nullptr and std::isfinite(*doubleValue) and
162 *doubleValue >= static_cast<double>(std::numeric_limits<float>::lowest()) and
163 *doubleValue <= static_cast<double>(std::numeric_limits<float>::max()) and
164 static_cast<double>(static_cast<float>(*doubleValue)) == *doubleValue;
165 case model::ColumnType::Double:
166 return doubleValue != nullptr and std::isfinite(*doubleValue);
167 case model::ColumnType::String:
168 return std::holds_alternative<std::string>(storedValue);
169 case model::ColumnType::Date:
170 return compatibleValue<Date>(storedValue);
171 case model::ColumnType::TimeOfDay:
172 return compatibleValue<TimeOfDay>(storedValue);
173 case model::ColumnType::LocalDateTime:
174 return compatibleValue<LocalDateTime>(storedValue);
175 case model::ColumnType::UtcTimestamp:
176 return compatibleValue<UtcTimestamp>(storedValue);
177 case model::ColumnType::Uuid:
178 return compatibleValue<Uuid>(storedValue);
179 }
180
181 return false;
182 }
