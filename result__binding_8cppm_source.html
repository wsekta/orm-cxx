172inline auto bindNull(soci::values& values, std::string_view name, model::ColumnType logicalType) -> void
173{
174 switch (logicalType)
175 {
176 case model::ColumnType::Bool:
177 case model::ColumnType::Char:
178 case model::ColumnType::UnsignedChar:
179 case model::ColumnType::Short:
180 case model::ColumnType::UnsignedShort:
181 case model::ColumnType::Int:
182 bindTypedNull(values, name, int{});
183 return;
184 case model::ColumnType::UnsignedInt:
185 case model::ColumnType::UnsignedLongLong:
186 bindTypedNull(values, name, static_cast<unsigned long long>(0));
187 return;
188 case model::ColumnType::LongLong:
189 bindTypedNull(values, name, static_cast<long long>(0));
190 return;
191 case model::ColumnType::Float:
192 case model::ColumnType::Double:
193 bindTypedNull(values, name, 0.0);
194 return;
195 case model::ColumnType::String:
196 bindTypedNull(values, name, std::string{});
197 return;
198 case model::ColumnType::Date:
199 case model::ColumnType::TimeOfDay:
200 case model::ColumnType::LocalDateTime:
201 case model::ColumnType::UtcTimestamp:
202 case model::ColumnType::Uuid:
203 bindTypedNull(values, name, std::string{});
204 return;
205 }
206
207 throw std::invalid_argument{"Cannot bind NULL parameter with unsupported column type"};
208}

220inline auto bindBoundValue(soci::values& values, std::string_view name, const BoundValue& value) -> void
221{
222 if (value.isNull())
223 {
224 bindNull(values, name, value.logicalType);
225 return;
226 }
227
228 if (not hasCompatibleStorage(value))
229 {
230 throw ConversionError{"Bound value storage does not match its logical column type"};
231 }
232
233 const auto parameterName = std::string{name};
234 std::visit([&values, &parameterName, columnType = value.logicalType](const auto& storedValue)
235 { setScalarFieldValue(values, parameterName, storedValue, columnType); }, value.value.value());
236}

238inline auto bindStatementParameter(soci::values& values, const StatementParameter& parameter) -> void
239{
240 bindBoundValue(values, parameter.name, parameter.getBoundValue());
