49// Keep this ordering identical to emitSqlFullFields, including the flattened
50// to-one aliases used by joined and non-joined hydration.
51inline auto modelResultColumns(model::ModelView model, bool joined) -> std::vector<ResultColumn>
52{
53 std::vector<ResultColumn> result;
54 for (const auto& column : model->columns)
55 {
56 if (column.kind == model::FieldKind::Scalar)
57 {
58 result.push_back({std::string{model->tableName} + "_" + std::string{column.name}, column.type.value()});
59 }
60 else
61 {
62 const auto target = model.resolveTarget(column);
63 if (!target)
64 throw ConversionError{"A related result model is not available in its schema"};
65 for (const auto& child : target->columns)
66 {
67 if (joined || child.isPrimaryKey)
68 {
69 const auto prefix = joined ? std::string{column.name} :
70 std::string{model->tableName} + "_" + std::string{column.name};
71 result.push_back({prefix + "_" + std::string{child.name}, child.type.value()});
72 }
73 }
74 }
75 }
76 return result;
77}

107template <typename Result, typename Projections>
108auto projectionResultColumns(model::ModelView model, const Projections& projections) -> std::vector<ResultColumn>
109{
110 std::vector<ResultColumn> result;
111 result.reserve(projections.size());
112 for (const auto& projection : projections)
113 {
114 if constexpr (requires { projection.alias; })
115 result.push_back({std::string{projection.alias}, resultSourceType(projection.source, model)});
116 else
117 {
118 const auto source = std::visit([](const auto& value) { return resultSource(value); }, projection.source);
119 result.push_back({projection.resultField, resultSourceType(source, model)});
120 }
121 [&]<std::size_t... Indices>(std::index_sequence<Indices...>)
122 {
123 constexpr auto fields = reflection::fields<Result>();
124 auto matchField = [&]<std::size_t Index>()
125 {

137/**
138 * Owns stable per-column buffers for explicit SOCI exchanges. New scalar types
139 * always use into(string, indicator), bypassing dynamic row/date conversion.
140 * Its row view retains the existing hydration and numeric checks without
141 * allocating unbound soci::values holders, whose lifetime SOCI cannot manage.
142 */

190 static auto storageFor(model::ColumnType type) -> Storage
191 {
192 using enum model::ColumnType;
193 switch (type)
194 {
195 case Bool:
196 case Char:
197 case UnsignedChar:
198 case Short:
199 case UnsignedShort:
200 case Int:
201 return int{};
202 case UnsignedInt:
203 case UnsignedLongLong:
204 return static_cast<unsigned long long>(0);
205 case LongLong:
206 return static_cast<long long>(0);
207 case Float:
208 case Double:
209 return double{};
210 case String:
211 case Date:
212 case TimeOfDay:
213 case LocalDateTime:
214 case UtcTimestamp:
215 case Uuid:
216 return std::string{};
217 }
218 throw ConversionError{"Unsupported result column type"};
219 }

256 template <typename T>
257 auto get(std::string_view alias) const -> T
258 {
259 const auto& buffer = find(alias);
260 if (buffer.indicator == soci::i_null)
261 throw ConversionError{"A non-null result field contains NULL"};
262 const auto* value = std::get_if<T>(&buffer.value);
263 if (!value)
264 throw ConversionError{"The result field has a different storage representation"};
265 return *value;
266 }

268 explicit TypedResultStorage(std::vector<ResultColumn> columns)
269 {
270 buffers.reserve(columns.size());
271 for (auto& column : columns)
272 {
273 if (column.requestedType &&
274 (isEncodedResultType(column.logicalType) || isEncodedResultType(*column.requestedType)) &&
275 column.logicalType != *column.requestedType)
276 throw ConversionError{"A projected scalar type does not match its requested result field"};
277 auto storage = storageFor(column.logicalType);
278 buffers.push_back(std::make_unique<Buffer>(Buffer{std::move(column), std::move(storage), soci::i_null}));
279 }
280 }

282 template <typename Consume, typename Validate = NoResultValidation>
283 auto read(soci::session& session, std::string_view sql, soci::values& parameters, Consume consume,
284 std::optional<BackendType> backend = std::nullopt, Validate validate = {}) -> void
285 {
286 const bool signedSqlIntegers =
287 backend && (*backend == BackendType::Postgres || *backend == BackendType::Sqlite);
288 // Preserve the SQL sign until checkedNumericCast runs; an unsigned
289 // SOCI exchange can turn -1 into the maximum unsigned value.
290 for (auto& buffer : buffers)
291 {
292 if (buffer->column.logicalType == model::ColumnType::UnsignedInt ||
293 buffer->column.logicalType == model::ColumnType::UnsignedLongLong)
294 buffer->value = signedSqlIntegers ? Storage{static_cast<long long>(0)} :
295 Storage{static_cast<unsigned long long>(0)};
296 }
297 soci::statement statement{session};
298 statement.alloc();
299 statement.prepare(std::string{sql});
300 statement.exchange(soci::use(parameters));
301 for (auto& buffer : buffers)
302 {
303 std::visit([&](auto& value) { statement.exchange(soci::into(value, buffer->indicator)); }, buffer->value);
304 }
305 statement.define_and_bind();
306 statement.execute();
307 while (statement.fetch())
308 {
309 for (const auto& buffer : buffers)
310 {
311 if (buffer->indicator != soci::i_ok && buffer->indicator != soci::i_null)
312 throw ConversionError{"A database result was truncated by its transport"};
313 if (buffer->indicator != soci::i_null && isEncodedResultType(buffer->column.logicalType))
314 {
315 auto value = decodedResult(buffer->column.logicalType, std::get<std::string>(buffer->value));
316 if (backend)
317 {
318 try
319 {
320 std::visit(
321 [&](const auto& scalar)
322 {
323 if constexpr (SociEncodedValue<std::remove_cvref_t<decltype(scalar)>>)
324 orm::detail::values::validateNative(scalar, *backend);
325 },
326 value);
327 }
328 catch (const std::invalid_argument&)
329 {
330 throw ConversionError{"A database result exceeds its native scalar type range"};
331 }
332 }
333 validate(BoundValue{.logicalType = buffer->column.logicalType, .value = std::move(value)});
334 }
335 }
336 consume(*this);
337 }
338 }
