27/**
28 * A session-bound, stable exchange buffer for one Binary16 parameter or result.
29 * The buffer and its indicator must outlive the statement using them. SQLite
30 * results additionally require typeof(value) from the same row, because SOCI's
31 * blob exchange converts other storage classes before returning the bytes.
32 */
33class Binary16Buffer
34{
35 soci::blob blob_;
36 soci::indicator indicator_{soci::i_null};
37
38public:
39 explicit Binary16Buffer(soci::session& session) : blob_{session} {}
40
41 auto set(const std::optional<orm::Uuid>& value) -> void
42 {
43 indicator_ = value ? soci::i_ok : soci::i_null;
44 blob_.trim(0);
45 if (value)
46 {
47 // SOCI's blob API takes `const char*`; `std::byte` carries the same
48 // representation and is the type the UUID codec exposes.
49 const auto* bytes = reinterpret_cast<const char*>(value->bytes().data());
50 if (blob_.write_from_start(bytes, value->bytes().size()) != value->bytes().size())
51 throw ConversionError{"Could not write a complete Binary16 value"};
52 }
53 }
54
55 auto exchangeUse(soci::statement& statement, const std::string& name) -> void;
56 auto exchangeInto(soci::statement& statement) -> void;
57
58 [[nodiscard]] auto indicator() const noexcept -> soci::indicator
59 {
60 return indicator_;
61 }
62
63 [[nodiscard]] auto read(BackendType backend, std::string_view sqliteStorageClass = {}) -> std::optional<orm::Uuid>
64 {
65 if (backend != BackendType::Sqlite && backend != BackendType::Mysql)
66 throw ConversionError{"Binary16 is unavailable for this backend"};
67 if (indicator_ == soci::i_null)
68 return std::nullopt;
69 if (indicator_ != soci::i_ok)
70 throw ConversionError{"A Binary16 result was truncated by its transport"};
71 if (backend == BackendType::Sqlite && sqliteStorageClass != "blob")
72 throw ConversionError{"A Binary16 result is not a SQLite BLOB"};
73 std::array<std::byte, 16> bytes{};
74 auto* buffer = reinterpret_cast<char*>(bytes.data());
75 if (blob_.get_len() != bytes.size() || blob_.read_from_start(buffer, bytes.size()) != bytes.size())
76 throw ConversionError{"A Binary16 result must contain exactly 16 bytes"};
77 return orm::Uuid{bytes};
78 }

193// Keep this ordering identical to emitSqlFullFields, including the flattened
194// to-one aliases used by joined and non-joined hydration.
195inline auto modelResultColumns(model::ModelView model, bool joined) -> std::vector<ResultColumn>
196{
197 std::vector<ResultColumn> result;
198 for (const auto& column : model->columns)
199 {
200 if (column.kind == model::FieldKind::Scalar)
201 {
202 result.push_back({std::string{model->tableName} + "_" + std::string{column.name}, column.type.value(),
203 std::nullopt, &column});
204 }
205 else
206 {
207 const auto target = model.resolveTarget(column);
208 if (!target)
209 throw ConversionError{"A related result model is not available in its schema"};
210 for (const auto& child : target->columns)
211 {
212 if (joined || child.isPrimaryKey)
213 {
214 const auto prefix = joined ? std::string{column.name} :
215 std::string{model->tableName} + "_" + std::string{column.name};
216 result.push_back(
217 {prefix + "_" + std::string{child.name}, child.type.value(), std::nullopt, &child});
218 }
219 }
220 }
221 }
222 return result;

253template <typename Result, typename Projections>
254auto projectionResultColumns(model::ModelView model, const Projections& projections) -> std::vector<ResultColumn>
255{
256 std::vector<ResultColumn> result;
257 result.reserve(projections.size());
258 for (const auto& projection : projections)
259 {
260 if constexpr (requires { projection.alias; })
261 {
262 result.push_back({std::string{projection.alias}, resultSourceType(projection.source, model)});
263 if (result.back().logicalType == model::ColumnType::Uuid)
264 result.back().column = db::detail::resolveSqlColumn(projection.source, model).column;
265 }
266 else
267 {
268 const auto source = std::visit([](const auto& value) { return resultSource(value); }, projection.source);
269 result.push_back({projection.resultField, resultSourceType(source, model)});
270 if (result.back().logicalType == model::ColumnType::Uuid)
271 result.back().column = db::detail::resolveSqlColumn(source, model).column;
272 }
273 [&]<std::size_t... Indices>(std::index_sequence<Indices...>)
274 {
275 constexpr auto fields = reflection::fields<Result>();
276 auto matchField = [&]<std::size_t Index>()
277 {
278 using field_t = optional_value_t<reflection::field_type_t<Result, Index>>;
279 if constexpr (requires { model::LogicalTypeTraits<field_t>::value; })
280 if (result.back().alias == fields[Index].name)
281 result.back().requestedType = model::LogicalTypeTraits<field_t>::value;
282 };
283 (matchField.template operator()<Indices>(), ...);
284 }(std::make_index_sequence<reflection::fieldCount<Result>>{});
285 }
286 return result;

325 {

339 static auto storageFor(model::ColumnType type) -> Storage
340 {
341 using enum model::ColumnType;
342 switch (type)
343 {
344 case Bool:
345 case Char:
346 case UnsignedChar:
347 case Short:
348 case UnsignedShort:
349 case Int:
350 return int{};
351 case UnsignedInt:
352 case UnsignedLongLong:
353 return static_cast<unsigned long long>(0);
354 case LongLong:
355 return static_cast<long long>(0);
356 case Float:
357 case Double:
358 return double{};
359 case String:
360 case Date:
361 case TimeOfDay:
362 case LocalDateTime:
363 case UtcTimestamp:
364 case Uuid:
365 return std::string{};
366 }
367 throw ConversionError{"Unsupported result column type"};
368 }

405 template <typename T>
406 auto get(std::string_view alias) const -> T
407 {
408 const auto& buffer = find(alias);
409 if (buffer.indicator == soci::i_null)
410 throw ConversionError{"A non-null result field contains NULL"};
411 const auto* value = std::get_if<T>(&buffer.value);
412 if (!value)
413 throw ConversionError{"The result field has a different storage representation"};
414 return *value;
415 }

417 explicit TypedResultStorage(std::vector<ResultColumn> columns)
418 {
419 buffers.reserve(columns.size());
420 for (auto& column : columns)
421 {
422 if (column.requestedType &&
423 (isEncodedResultType(column.logicalType) || isEncodedResultType(*column.requestedType)) &&
424 column.logicalType != *column.requestedType)
425 throw ConversionError{"A projected scalar type does not match its requested result field"};
426 auto storage = storageFor(column.logicalType);
427 buffers.push_back(
428 std::make_unique<Buffer>(Buffer{std::move(column), std::move(storage), soci::i_null, nullptr, {}}));
429 }

432 template <typename Consume, typename Validate = NoResultValidation>
433 auto read(soci::session& session, std::string_view sql, StatementBindings& parameters, Consume consume,
434 std::optional<BackendType> backend = std::nullopt, Validate validate = {}) -> void
435 {
436 const bool signedSqlIntegers =
437 backend && (*backend == BackendType::Postgres || *backend == BackendType::Sqlite);
438 // Preserve the SQL sign until checkedNumericCast runs; an unsigned
439 // SOCI exchange can turn -1 into the maximum unsigned value.
440 for (auto& buffer : buffers)
441 {
442 buffer->binary.reset();
443 buffer->sqliteStorageClass.clear();
444 if (buffer->column.logicalType == model::ColumnType::UnsignedInt ||
445 buffer->column.logicalType == model::ColumnType::UnsignedLongLong)
446 buffer->value = signedSqlIntegers ? Storage{static_cast<long long>(0)} :
447 Storage{static_cast<unsigned long long>(0)};
448 if (backend && buffer->column.column && buffer->column.logicalType == model::ColumnType::Uuid &&
449 buffer->column.column->uuidStorage(*backend) == model::UuidStorage::Binary16)
450 buffer->binary = std::make_unique<Binary16Buffer>(session);
451 }
452 soci::statement statement{session};
453 statement.alloc();
454 statement.prepare(std::string{sql});
455 parameters.exchange(statement);
456 for (auto& buffer : buffers)
457 {
458 if (buffer->binary)
459 buffer->binary->exchangeInto(statement);
460 else
461 std::visit([&](auto& value) { statement.exchange(soci::into(value, buffer->indicator)); },
462 buffer->value);
463 }
464 if (backend && *backend == BackendType::Sqlite)
465 for (auto& buffer : buffers)
466 if (buffer->binary || (buffer->column.column && buffer->column.logicalType == model::ColumnType::Uuid &&
467 buffer->column.column->uuidStorage(*backend) == model::UuidStorage::Text36))
468 statement.exchange(soci::into(buffer->sqliteStorageClass));
469 statement.define_and_bind();
470 statement.execute();
471 while (statement.fetch())
472 {
473 for (const auto& buffer : buffers)
474 {
475 // A Text36 column stores a canonical TEXT UUID, so the native UUID
476 // validation (Postgres/MariaDB only) must not apply to its result.
477 const bool text36Result = backend && buffer->column.logicalType == model::ColumnType::Uuid &&
478 buffer->column.column != nullptr &&
479 buffer->column.column->uuidStorage(*backend) == model::UuidStorage::Text36;
480 if (buffer->binary)
481 {
482 const auto value = buffer->binary->read(*backend, buffer->sqliteStorageClass);
483 buffer->indicator = buffer->binary->indicator();
484 if (value)
485 buffer->value = value->toString();
486 }
487 else if (backend && *backend == BackendType::Sqlite &&
488 buffer->column.logicalType == model::ColumnType::Uuid && buffer->column.column != nullptr &&
489 buffer->column.column->uuidStorage(*backend) == model::UuidStorage::Text36 &&
490 buffer->indicator != soci::i_null && buffer->sqliteStorageClass != "text")
491 {
492 throw ConversionError{"A Text36 result is not a SQLite TEXT value"};
493 }
494 validateResultIndicator(buffer->indicator);
495 if (buffer->indicator != soci::i_null && isEncodedResultType(buffer->column.logicalType))
496 {
497 auto value = decodedResult(buffer->column.logicalType, std::get<std::string>(buffer->value));
498 if (backend && !buffer->binary && !text36Result)
499 {
500 try
501 {
502 std::visit(
503 [&](const auto& scalar)
504 {
505 if constexpr (SociEncodedValue<std::remove_cvref_t<decltype(scalar)>>)
506 orm::detail::values::validateNative(scalar, *backend);
507 },
508 value);
509 }
510 catch (const std::invalid_argument&)
511 {
512 throw ConversionError{"A database result exceeds its native scalar type range"};
513 }
514 }
515 validate(BoundValue{.logicalType = buffer->column.logicalType,
516 .value = std::move(value),
517 .uuidStorage = buffer->binary ?
518 model::UuidStorage::Binary16 :
519 (backend && buffer->column.column != nullptr &&
520 buffer->column.column->uuidStorage(*backend) ==
521 model::UuidStorage::Text36 ?
522 model::UuidStorage::Text36 :
523 model::UuidStorage::Native)});
524 }
525 }
526 consume(*this);
527 }
528 }
