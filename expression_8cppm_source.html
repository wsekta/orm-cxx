188template <typename S, ComparisonOperator Op>
189struct ComparisonMeta : ExpressionMeta<typename S::Model, S::writeSafe, S::containsCollection>
190{
191 using Source = S;
192 using Dynamic = std::conditional_t<S::isAggregate, TypedAggregatePredicate<typename S::Model>,
193 TypedPredicate<typename S::Model, S::writeSafe, S::containsCollection>>;
194 inline static constexpr auto operation = Op;
195 inline static constexpr bool isPredicate = !S::isAggregate;
196 inline static constexpr bool isAggregatePredicate = S::isAggregate;
197 inline static constexpr bool staticSqlEligible = detail::staticSqlEligible<S>;
198 template <typename Children, typename Args>
199 static auto render(const Children& children, const Args& args)
200 {
201 const auto source = TypedAccess::erase(std::get<0>(children), args);
202 const auto value = typedValue<typename S::Value>(resolveValue(std::get<1>(children), args));
203 if constexpr (S::isAggregate)
204 {
205 if constexpr (Op == ComparisonOperator::Equal)
206 return source == value;
207 else if constexpr (Op == ComparisonOperator::NotEqual)
208 return source != value;
209 else if constexpr (Op == ComparisonOperator::Greater)
210 return source > value;
211 else if constexpr (Op == ComparisonOperator::GreaterOrEqual)
212 return source >= value;
213 else if constexpr (Op == ComparisonOperator::Less)
214 return source < value;
215 else
216 return source <= value;
217 }
218 else
219 return Predicate{PredicateNode{ComparisonExpression{source, Op, value}}};
220 }
221 template <typename Children, typename Args, typename Callback>
222 static auto visit(const Children& children, const Args& args, Callback& callback) -> void
223 {
224 callback(typedValue<typename S::Value>(resolveValue(std::get<1>(children), args)));
225 }
226};

297template <typename S, ListOperator Op, typename Stored>
298struct ListMeta : ExpressionMeta<typename S::Model, S::writeSafe, S::containsCollection>
299{
300 using Source = S;
301 using Values = Stored;
302 using Dynamic = TypedPredicate<typename S::Model, S::writeSafe, S::containsCollection>;
303 inline static constexpr auto operation = Op;
304 inline static constexpr bool isFixedList = ListTraits<Stored>::fixed;
305 inline static constexpr auto arity = ListTraits<Stored>::size;
306 inline static constexpr bool isPredicate = true;
307 inline static constexpr bool staticSqlEligible = detail::staticSqlEligible<S> && isFixedList;
308 static_assert(!isFixedList || arity != 0, "ORM_QUERY_EMPTY_IN: IN requires at least one value");
309 template <typename Children, typename Args, typename Callback>
310 static auto visit(const Children& children, const Args& args, Callback& callback) -> void
311 {
312 const auto& stored = std::get<1>(children);
313 if constexpr (requires { stored.values; })
314 std::apply([&](const auto&... value)
315 { (callback(typedValue<typename S::Value>(resolveValue(value, args))), ...); }, stored.values);
316 else
317 {
318 const auto& values = resolveValue(stored, args);
319 if (values.empty())
320 throw std::invalid_argument{"IN predicate requires at least one value"};
321 for (const auto& value : values)
322 callback(typedValue<typename S::Value>(
323 static_cast<typename std::remove_cvref_t<decltype(values)>::value_type>(value)));
324 }
325 }

331 visit(children, args, append);
332 return Predicate{
333 PredicateNode{ListExpression{TypedAccess::erase(std::get<0>(children), args), Op, std::move(values)}}};

336template <ListOperator Op, typename S, typename... V>
337constexpr auto fixedList(const S& source, V&&... value)
338{
339 auto values = FixedValues<decltype(captureValue(std::forward<V>(value)))...>{
340 std::tuple{captureValue(std::forward<V>(value))...}};

343template <ListOperator Op, typename S, typename Container>
344constexpr auto containerList(const S& source, const Container& values)
345{
346 using C = std::remove_cvref_t<Container>;
347 if constexpr (isParameter<C>)
348 return makeExpression<ExprKind::List, ListMeta<S, Op, C>>(source, values);
349 else if constexpr (ContainerTraits<C>::fixed)
350 return [&]<std::size_t... I>(std::index_sequence<I...>)
351 { return fixedList<Op>(source, values[I]...); }(std::make_index_sequence<ContainerTraits<C>::size>{});
352 else
353 {
354 if (values.size() == 0)
355 throw std::invalid_argument{"IN predicate requires at least one value"};
356 using V = decltype(captureValue(std::declval<typename C::value_type>()));
357 std::vector<V> owned;
358 owned.reserve(values.size());
359 for (const auto& value : values)
360 owned.push_back(captureValue(static_cast<typename C::value_type>(value)));
361 return makeExpression<ExprKind::List, ListMeta<S, Op, decltype(owned)>>(source, std::move(owned));
