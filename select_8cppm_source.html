212 /**
213 * @brief Explicitly loads a mapped OneToMany or ManyToMany collection.
214 *
215 * Repeating the same
216 * include is idempotent. Collection loading is performed
217 * after the root SELECT, so root pagination is
218 * preserved.
219 */
220 template <auto Member>
221// requires query::detail::ORM_QUERY_COLLECTION<Member> &&
222// query::detail::ORM_QUERY_MODEL_TYPE<typename query::detail::CollectionTraits<Member>::Model, T>
223 auto include() -> Query<T>&
224 {
225 const auto relation = query::detail::CollectionTraits<Member>::name();
226 if (std::ranges::find(data.includes, relation) == data.includes.end())
227 data.includes.push_back(relation);
228 return *this;
229 }
