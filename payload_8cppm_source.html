123template <typename T>
124auto toPrimaryKeyValue(const T& value, const std::string& fieldName) -> query::QueryValue
125{
126 using value_t = std::remove_cv_t<T>;
127
128 if constexpr (isOptional<value_t>)
129 {
130 if (not value.has_value())
131 {
132 throw std::invalid_argument{"Primary-key field must not be NULL: " + fieldName};
133 }
134
135 return toPrimaryKeyValue(value.value(), fieldName);
136 }
137 else if constexpr (requires { query::QueryValue{value}; })
138 {
139 if constexpr (SociEncodedValue<value_t>)
140 (void)encodeScalarValue(value);
141 return query::QueryValue{value};
142 }
143 else
144 {
145 throw std::invalid_argument{"Unsupported primary-key field type: " + fieldName};
146 }

149template <typename SchemaType, typename T>
150auto getPrimaryKey(const T& object) -> PrimaryKey
151{
152 static_assert(SchemaType::template contains<std::remove_cv_t<T>>,
153 "A primary key can be read only from a model in the selected Schema");
154 constexpr auto descriptor = model::modelView<SchemaType, std::remove_cv_t<T>>();
155 const auto objectAsTuple = reflection::fieldPointers(object);
156 PrimaryKey key;
157 key.reserve(descriptor->primaryKeyIndices.size());
158
159 auto appendPrimaryKeyField = [&key]<typename Index>(Index, const auto* field)
160 {
161 using field_t = std::decay_t<decltype(*field)>;
162
163 if constexpr (orm::is_relation_collection_v<field_t>)
164 {
165 return;
166 }
167 else
168 {
169 constexpr auto* column = detail::mappedColumn<std::remove_cv_t<T>, SchemaType, Index::value>;
170 if constexpr (column != nullptr)
171 {
172 if (column->isPrimaryKey)
173 {
174 key.push_back(toPrimaryKeyValue(*field, std::string{column->fieldName}));
175 }
176 }
177 }
178 };
179
180 utils::constexpr_for_tuple(objectAsTuple, appendPrimaryKeyField);
181
182 if (key.empty())
183 {
184 throw std::invalid_argument{"Relation endpoint must define a non-empty primary key"};
185 }
186
187 return key;
188}

190inline auto getPrimaryKeyValue(const auto& values, const std::string& name, model::ColumnType type) -> query::QueryValue
191{
192 if (values.get_indicator(name) == soci::i_null)
193 {
194 throw ConversionError{"Cannot hydrate NULL relation primary key: " + name};
195 }
196
197 const auto fromStorage = [type, &name](query::QueryValue::Value value)
198 {
199 try
200 {
201 return query::QueryValue::fromStorage(type, std::move(value));
202 }
203 catch (const std::invalid_argument&)
204 {
205 throw ConversionError{"Cannot hydrate relation primary key without data loss: " + name};
206 }
207 };
208
209 switch (type)
210 {
211 case model::ColumnType::Bool:
212 case model::ColumnType::Char:
213 case model::ColumnType::UnsignedChar:
214 case model::ColumnType::Short:
215 case model::ColumnType::UnsignedShort:
216 case model::ColumnType::Int:
217 return fromStorage(getNumericValue<int>(values, name));
218 case model::ColumnType::UnsignedInt:
219 case model::ColumnType::UnsignedLongLong:
220 return fromStorage(getNumericValue<unsigned long long>(values, name));
221 case model::ColumnType::LongLong:
222 return fromStorage(getNumericValue<long long>(values, name));
223 case model::ColumnType::Float:
224 case model::ColumnType::Double:
225 return fromStorage(getNumericValue<double>(values, name));
226 case model::ColumnType::String:
227 return fromStorage(values.template get<std::string>(name));
228 case model::ColumnType::Date:
229 return fromStorage(getEncodedScalarValue<Date>(values, name));
230 case model::ColumnType::TimeOfDay:
231 return fromStorage(getEncodedScalarValue<TimeOfDay>(values, name));

243inline auto getPrimaryKeyColumns(const model::ModelView& model) -> std::vector<const model::ColumnView*>
244{
245 std::vector<const model::ColumnView*> columns;
246 columns.reserve(model->primaryKeyIndices.size());
247
248 for (const auto columnIndex : model->primaryKeyIndices)
249 {
250 if (columnIndex >= model->columns.size())
251 {
252 throw std::invalid_argument{"Primary-key descriptor points outside the model columns"};
253 }
254
255 const auto& column = model->columns[columnIndex];
256 if (column.kind != model::FieldKind::Scalar)
257 {
258 throw std::invalid_argument{"Relations with model-valued primary-key fields are not supported"};
259 }
260 columns.push_back(&column);
261 }
262
263 if (columns.empty())
264 {
265 throw std::invalid_argument{"Relation endpoint must define a non-empty primary key"};
266 }
267
268 return columns;
269}
