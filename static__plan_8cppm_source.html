257 template <typename... Args>
258 auto toDynamic(Args&&... values) const
259 {
260 validateShape();
261 detail::validateParameters<StaticPlan, Args...>();
262 const auto args = std::forward_as_tuple(values...);
263 if constexpr (Operation == detail::PlanOperation::Remove)
264 return detail::bindExpression(std::get<0>(state), args);
265 else if constexpr (Operation == detail::PlanOperation::Update)
266 {
267 orm::Update<Model> result;
268 std::apply([&](const auto&... assignment) { (detail::applyAssignment(result, assignment, args), ...); },
269 std::get<5>(state));
270 result.where(detail::bindExpression(std::get<0>(state), args));
271 return result;
272 }
273 else
274 {
275 auto result = [&]
276 {
277 if constexpr (isProjection)
278 {
279 orm::ProjectionQuery<Model, Result> output;
280 std::apply([&](const auto&... projection) { output.project(projection...); },
281 std::get<4>(state));
282 return output;
283 }
284 else
285 return orm::Query<Model>{};
286 }();
287 if constexpr (hasPredicate)
288 result.where(detail::bindExpression(std::get<0>(state), args));
289 std::apply([&](const auto&... order) { result.orderBy(order...); }, std::get<1>(state));
290 std::apply([&](const auto&... group) { result.groupBy(group...); }, std::get<2>(state));
291 if constexpr (hasHaving)
292 result.having(detail::bindExpression(std::get<3>(state), args));
293 if constexpr (Distinct)
294 result.distinct();
295 if constexpr (!Join)
296 result.disableJoining();
297 if constexpr (hasLimit)
298 result.limit(paginationValue(std::get<6>(state), args));
299 if constexpr (hasOffset)
300 result.offset(paginationValue(std::get<7>(state), args));
301 if constexpr (!isProjection)
302 std::apply([&](const auto&... include)
303 { (result.template include<std::remove_cvref_t<decltype(include)>::member>(), ...); },
304 std::get<8>(state));
305 return result;
306 }
