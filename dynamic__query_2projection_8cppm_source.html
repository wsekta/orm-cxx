29 switch (type)

61template <typename T>
62consteval auto projectionLogicalType() -> std::optional<model::ColumnType>
63{
64 using field_t = std::remove_cvref_t<T>;
65 using value_t = std::remove_cv_t<model::detail::static_optional_value_t<field_t>>;
66 if constexpr (requires { model::LogicalTypeTraits<value_t>::value; })
67 {
68 return model::LogicalTypeTraits<value_t>::value;
69 }
70 else
71 {
72 return std::nullopt;
73 }

85template <typename Projections>
86inline auto validateProjectionAliasNames(const Projections& projections,
87 const std::vector<ProjectionResultField>& fields) -> void
88{
89 if (projections.empty())
90 {
91 throw std::invalid_argument{"Projection query requires at least one projected field"};
92 }
93
94 std::vector<std::string> resultFields;
95 resultFields.reserve(fields.size());
96
97 for (const auto& field : fields)
98 {
99 if (not field.type.has_value() or not isSupportedProjectionResultType(field.type.value()))
100 {
101 throw std::invalid_argument{"Unsupported projection result field type: " + field.name};
102 }
103
104 resultFields.push_back(field.name);
105 }
106
107 std::vector<std::string> projectedFields;
108 projectedFields.reserve(projections.size());
109
110 for (const auto& projection : projections)
111 {
112 const std::string alias{projectionAlias(projection)};
113 if (alias.empty())
114 {
115 throw std::invalid_argument{"Projection alias must not be empty"};
116 }
117
118 if (std::find(resultFields.begin(), resultFields.end(), alias) == resultFields.end())
119 {
120 throw std::invalid_argument{"Projection alias does not match a result field: " + alias};
121 }
122
123 if (std::find(projectedFields.begin(), projectedFields.end(), alias) != projectedFields.end())
124 {
125 throw std::invalid_argument{"Duplicate projection alias: " + alias};
126 }
127
128 projectedFields.push_back(alias);
129 }
130
131 for (const auto& resultField : resultFields)
132 {
133 if (std::find(projectedFields.begin(), projectedFields.end(), resultField) == projectedFields.end())
134 {
135 throw std::invalid_argument{"Missing projection alias for result field: " + resultField};
136 }
137 }
138}

140template <typename Result, typename Projections>
141auto validateProjectionAliases(const Projections& projections) -> void
142{
143 std::vector<ProjectionResultField> resultFields;
144 resultFields.reserve(reflection::fieldCount<Result>);
145
146 [&]<std::size_t... Indices>(std::index_sequence<Indices...>)
147 {
148 constexpr auto fields = reflection::fields<Result>();
149 auto appendField = [&]<std::size_t Index>()
150 {
151 constexpr auto type = projectionLogicalType<reflection::field_type_t<Result, Index>>();
152 resultFields.emplace_back(std::string{fields[Index].name}, type);
153 };
154 (appendField.template operator()<Indices>(), ...);
155 }(std::make_index_sequence<reflection::fieldCount<Result>>{});
156
157 validateProjectionAliasNames(projections, resultFields);
158}
