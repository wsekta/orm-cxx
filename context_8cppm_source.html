227 template <typename Plan, typename... Args>
228 auto selectPlanImpl(const Plan& plan, Args&&... args) -> std::vector<typename Plan::Result>
229 {
230 Plan::validateShape();
231 query::detail::validateParameters<Plan, Args...>();
232 if constexpr (query::detail::compiledSqlEligible<Plan>)
233 {
234 const auto values = std::forward_as_tuple(args...);
235 const auto flavor = database_->getBackend().compiledSqlFlavor();
236 if constexpr (
237 query::detail::compiledStatement<SchemaType, Plan, db::CompiledSqlFlavor::MySQLFamily>.valid)
238 {
239 if (flavor == db::CompiledSqlFlavor::MySQLFamily)
240 return executeCompiledPlan<Plan, db::CompiledSqlFlavor::MySQLFamily>(plan, values);
241 }
242 if constexpr (query::detail::compiledStatement<SchemaType, Plan, db::CompiledSqlFlavor::SQLite>.valid)
243 {
244 if (flavor == db::CompiledSqlFlavor::SQLite)
245 return executeCompiledPlan<Plan, db::CompiledSqlFlavor::SQLite>(plan, values);
246 }
247 if constexpr (
248 query::detail::compiledStatement<SchemaType, Plan, db::CompiledSqlFlavor::PostgreSQL>.valid)
249 {
250 if (flavor == db::CompiledSqlFlavor::PostgreSQL)
251 return executeCompiledPlan<Plan, db::CompiledSqlFlavor::PostgreSQL>(plan, values);
252 }
253 }
254 return executeDynamicPlan<>(plan, std::forward<Args>(args)...);

315 template <typename Plan, typename... Args>
316 auto executeDynamicPlan(const Plan& plan, Args&&... args)
317 {
318 auto bound = plan.toDynamic(std::forward<Args>(args)...);
319 if constexpr (Plan::operation == query::detail::PlanOperation::Select)
320 return selectImpl<>(bound);
321 else if constexpr (Plan::operation == query::detail::PlanOperation::Update)
322 return updateImpl<>(bound);
323 else
324 return removeImpl<typename Plan::Model>(query::detail::erase(bound));
325 }

327 template <typename Plan, db::CompiledSqlFlavor Flavor, typename Args>
328 auto executeCompiledPlan(const Plan& plan, const Args& values)
329 {
330 constexpr auto descriptor = model::modelView<SchemaType, typename Plan::Model>();
331 constexpr auto& compiled = query::detail::compiledStatement<SchemaType, Plan, Flavor>;
332 const auto requirements = compiled.program.view();
333 if constexpr (Plan::operation == query::detail::PlanOperation::Select)
334 {
335 if constexpr (Plan::isProjection)
336 detail::validateProjectionAliases<typename Plan::Result>(requirements.projections);
337 database_->ensureQuerySupported(descriptor, requirements);
338 const auto parameters = query::detail::collectParameters(plan, values);
339 auto resultColumns = [&]
340 {
341 if constexpr (Plan::isProjection)
342 return db::binding::projectionResultColumns<typename Plan::Result>(descriptor,
343 requirements.projections);
344 else
345 return db::binding::modelResultColumns(descriptor, Plan::shouldJoin);
346 }();
347 return executeSelectStatement<typename Plan::Model, typename Plan::Result, Plan::isProjection>(
348 db::StatementView{compiled.view(), parameters}, Plan::shouldJoin, std::move(resultColumns),
349 [&](auto& rows)
350 {
351 (void)rows;
352 if constexpr (!Plan::isProjection)
353 loadIncludedCollections<>(descriptor, requirements, rows);
354 });
355 }
356 else
357 {
358 constexpr auto operation =
359 Plan::operation == query::detail::PlanOperation::Update ? "update" : "remove";
360 database_->ensureModelSupported(descriptor, operation);
361 if constexpr (Plan::operation == query::detail::PlanOperation::Update)
362 database_->requireCapability(database_->getBackendCapabilities().mutations.update, operation,
363 "update is not supported");
364 else
365 database_->requireCapability(database_->getBackendCapabilities().mutations.remove, operation,
366 "remove is not supported");
367 database_->ensurePredicateSupported(
368 std::ranges::any_of(requirements.nodes, [](const auto& node)
369 { return node.kind == db::detail::SqlNodeKind::Collection; }),
370 operation);
371 database_->ensureQueryTypesSupported(descriptor, requirements, operation);
372 const auto parameters = query::detail::collectParameters(plan, values);
373 if constexpr (Plan::operation == query::detail::PlanOperation::Update)
374 for (const auto& assignment : requirements.assignments)
375 database_->ensureStringValueSupported(
376 *db::detail::resolveSqlColumn(assignment.source, descriptor, true).column,
377 parameters[assignment.parameter].value, operation);
378 return database_->executeMutation(db::StatementView{compiled.view(), parameters}, operation);
379 }
380 }
381 template <typename T>
382 auto selectImpl(Query<T>& query) -> std::vector<T>
383 {
384 constexpr auto descriptor = model::modelView<SchemaType, T>();
385 database_->ensureQuerySupported(descriptor, query.getData());
386 const auto statement = database_->getCommandGenerator().select(descriptor, query.getData());
387 return executeSelectStatement<T, T, false>(
388 db::StatementView{statement.sql, statement.parameters}, query.getData().shouldJoin,
389 db::binding::modelResultColumns(descriptor, query.getData().shouldJoin),
390 [&](auto& rows) { loadIncludedCollections<>(descriptor, query.getData(), rows); });
391 }

404 template <typename Source, typename Result, bool Projection, typename Loader>
405 auto executeSelectStatement(db::StatementView statement, bool shouldJoin,
406 std::vector<db::binding::ResultColumn> resultColumns,
407 Loader loadIncludes) -> std::vector<Result>
408 {
409 constexpr auto operation = Projection ? "select projection" : "select";
410 database_->ensureStatementWithinBindLimit(statement.parameters.size(), operation);
411 std::vector<Result> result;
412 try
413 {
414 std::optional<db::binding::TypedResultStorage> storage;
415 if (db::binding::hasEncodedResult(resultColumns))
416 storage.emplace(std::move(resultColumns));
417 soci::values parameterValues;
418 for (const auto& parameter : statement.parameters)
419 database_->ensureBoundValueSupported(parameter.getBoundValue(), operation);
420 detail::bindStatementParameters(database_->getBackend().runtime(), parameterValues,
421 statement.parameters);
422 if constexpr (Projection)
423 {
424 if (storage)
425 {
426 storage->read(
427 database_->sql, statement.sql, parameterValues,
428 [&](const auto& row)
429 {
430 db::binding::ProjectionPayload<Result> payload{};
431 soci::type_conversion<db::binding::ProjectionPayload<Result>>::from_base(
432 row, soci::i_ok, payload);
433 result.push_back(std::move(payload.value));
434 },
435 database_->backendType, [&](const db::BoundValue& value)
436 { database_->ensureBoundValueSupported(value, operation); });
437 }
438 else
439 {
440 soci::rowset<db::binding::ProjectionPayload<Result>> rows =
441 (database_->sql.prepare << statement.sql, soci::use(parameterValues));
442 for (auto& payload : rows)
443 result.push_back(std::move(payload.value));
444 }
445 }
446 else
447 {
448 auto readRows = [&]<bool Joined>()
449 {
450 if (storage)
451 {
452 storage->read(
453 database_->sql, statement.sql, parameterValues,
454 [&](const auto& row)
455 {
456 db::binding::BindingPayload<Source, SchemaType, Joined> payload{};
457 soci::type_conversion<
458 db::binding::BindingPayload<Source, SchemaType, Joined>>::from_base(row,
459 soci::i_ok,
460 payload);
461 result.push_back(std::move(payload.value));
462 },
463 database_->backendType, [&](const db::BoundValue& value)
464 { database_->ensureBoundValueSupported(value, operation); });
465 }
466 else
467 {
468 soci::rowset<db::binding::BindingPayload<Source, SchemaType, Joined>> rows =
469 (database_->sql.prepare << statement.sql, soci::use(parameterValues));
470 for (auto& payload : rows)
471 result.push_back(std::move(payload.value));
472 }
473 };
474 if (shouldJoin)
475 readRows.template operator()<true>();
476 else
477 readRows.template operator()<false>();
478 }
479 loadIncludes(result);
480 }
481 catch (const db::binding::ConversionError&)
482 {
483 throw DatabaseError{DatabaseErrorCode::Conversion, database_->backendType, operation,
484 Projection ? "A database result cannot be represented by the requested projection" :
485 "A database result cannot be represented by the requested model"};
486 }
487 catch (const soci::soci_error& error)
488 {
489 database_->throwTranslatedError(error, DatabaseErrorCode::Statement, operation);
490 }
491 return result;
492 }

511 *
512 * @tparam T The type of the query.

514 */

516 auto insertImpl(T object) -> void
517 {

519 database_->ensureModelSupported(model, "insert");
520 database_->requireCapability(database_->getBackendCapabilities().mutations.insert, "insert",
521 "insert is not supported");
522 const auto command = database_->getCommandGenerator().insert(model);
523
524 const db::binding::BindingPayload<T, SchemaType> payload{};

533 indicator);

535 soci::values parameterValues;

563 *
564 * @tparam T The type of the query model.

570 {

572 database_->ensureModelSupported(model, "update");
573 database_->requireCapability(database_->getBackendCapabilities().mutations.update, "update",
574 "update is not supported");

577 database_->ensurePredicateSupported(model, *update.getData().predicate, "update");
578 }

580 database_->ensureStringValueSupported(
581 *db::detail::resolveSqlColumn(db::detail::parseSqlSource(assignment.column.getPath()), model, true)
582 .column,
583 assignment.value.value, "update");
584 const auto statement = database_->getCommandGenerator().update(model, update.getData());
585
586 return database_->executeMutation(statement, "update");

695 /**
696 * @brief Creates or changes a OneToMany/ManyToMany association.
697 * @return 1 when database state changed, otherwise 0.
698 */
699 template <typename Owner, typename Target>
700 auto linkImpl(const Owner& owner, std::string_view relationField, const Target& target) -> std::size_t
701 {
702 constexpr auto ownerDescriptor = model::modelView<SchemaType, Owner>();
703 const auto* relation = detail::requireCollectionRelation(ownerDescriptor, relationField);
704 const auto targetDescriptor =
705 detail::requireCollectionTarget(ownerDescriptor, *relation, model::typeId<Target>());
706
707 database_->ensureModelSupported(ownerDescriptor, "link relation");
708 database_->ensureModelSupported(*targetDescriptor, "link relation");
709 const auto ownerKey = db::binding::getPrimaryKey<SchemaType>(owner);
710 const auto targetKey = db::binding::getPrimaryKey<SchemaType>(target);
711
712 if (relation->kind == model::RelationKind::OneToMany)
713 {
714 database_->requireCapability(database_->getBackendCapabilities().relations.oneToMany, "link relation",
715 "one-to-many relations are not supported");
716 }
717 else
718 {
719 database_->requireCapability(database_->getBackendCapabilities().relations.manyToMany, "link relation",
720 "many-to-many relations are not supported");
721 database_->requireCapability(database_->getBackendCapabilities().mutations.atomicInsertIfAbsent,
722 "link relation", "idempotent relation links are not supported");
723 }
724 database_->requireCapability(relation->kind == model::RelationKind::OneToMany ?
725 database_->getBackendCapabilities().mutations.update :
726 database_->getBackendCapabilities().mutations.insert,
727 "link relation", "relation mutations are not supported");
728
729 if (not database_->relationEndpointExists(ownerDescriptor, ownerKey) or
730 not database_->relationEndpointExists(*targetDescriptor, targetKey))
731 {
732 throw std::invalid_argument{"Cannot link relation endpoints that do not exist"};
733 }
734
735 const auto statement = db::relations::linkStatement(database_->getBackend().dialect(), ownerDescriptor,
736 *relation, ownerKey, targetKey);
737 if (relation->kind == model::RelationKind::ManyToMany)
738 return database_->executeInsertIfAbsent(
739 statement, db::relations::linkInsertSpec(ownerDescriptor, *relation, ownerKey, targetKey),
740 "link relation");
741 return database_->executeMutation(statement, "link relation");
742 }

746 * @return 1 when database state changed, otherwise 0.
747 */

749 auto unlinkImpl(const Owner& owner, std::string_view relationField, const Target& target) -> std::size_t
750 {
751 constexpr auto ownerDescriptor = model::modelView<SchemaType, Owner>();
752 const auto* relation = detail::requireCollectionRelation(ownerDescriptor, relationField);
753 const auto targetDescriptor =
754 detail::requireCollectionTarget(ownerDescriptor, *relation, model::typeId<Target>());
755
756 database_->ensureModelSupported(ownerDescriptor, "unlink relation");
757 database_->ensureModelSupported(*targetDescriptor, "unlink relation");
758 database_->requireCapability(relation->kind == model::RelationKind::OneToMany ?
759 database_->getBackendCapabilities().relations.oneToMany :
760 database_->getBackendCapabilities().relations.manyToMany,
761 "unlink relation", "the requested collection relation is not supported");
762 database_->requireCapability(relation->kind == model::RelationKind::OneToMany ?
763 database_->getBackendCapabilities().mutations.update :
764 database_->getBackendCapabilities().mutations.remove,
765 "unlink relation", "relation mutations are not supported");
766
767 const auto ownerKey = db::binding::getPrimaryKey<SchemaType>(owner);
768 const auto targetKey = db::binding::getPrimaryKey<SchemaType>(target);
769 database_->ensureEndpointKeySupported(ownerDescriptor, ownerKey, "unlink relation");
770 database_->ensureEndpointKeySupported(*targetDescriptor, targetKey, "unlink relation");
771 return database_->executeMutation(db::relations::unlinkStatement(database_->getBackend().dialect(),
772 ownerDescriptor, *relation, ownerKey,
773 targetKey),
774 "unlink relation");

777 template <typename Owner, typename Target, bool JoinedValues>
778 auto appendCollectionRows(
779 const db::Statement& statement,
780 std::map<db::binding::PrimaryKey, std::vector<Target>, db::binding::PrimaryKeyLess>& groupedTargets) -> void
781 {
782 database_->ensureStatementWithinBindLimit(statement.parameters.size(), "include collection");
783 soci::values parameterValues;
784 for (const auto& parameter : statement.parameters)
785 database_->ensureBoundValueSupported(parameter.getBoundValue(), "include collection");
786 detail::bindStatementParameters(database_->getBackend().runtime(), parameterValues, statement.parameters);
787 constexpr auto owner = model::modelView<SchemaType, Owner>();
788 std::vector<db::binding::ResultColumn> columns;
789 for (const auto* column : db::binding::getPrimaryKeyColumns(owner))
790 columns.push_back({db::binding::relationOwnerAlias(*column), column->type.value()});
791 auto targetColumns = db::binding::modelResultColumns(model::modelView<SchemaType, Target>(), JoinedValues);
792 for (auto& column : targetColumns)
793 columns.push_back(std::move(column));
794 if (db::binding::hasEncodedResult(columns))
795 {
796 db::binding::TypedResultStorage storage{std::move(columns)};
797 storage.read(
798 database_->sql, statement.sql, parameterValues,
799 [&](const auto& row)
800 {
801 db::binding::CollectionPayload<Owner, Target, SchemaType, JoinedValues> payload{};
802 soci::type_conversion<db::binding::CollectionPayload<Owner, Target, SchemaType, JoinedValues>>::
803 from_base(row, soci::i_ok, payload);
804 groupedTargets[payload.ownerKey].push_back(std::move(payload.value));
805 },
806 database_->backendType, [&](const db::BoundValue& value)
807 { database_->ensureBoundValueSupported(value, "include collection"); });
808 return;
809 }
810 soci::rowset<db::binding::CollectionPayload<Owner, Target, SchemaType, JoinedValues>> preparedRowSet =
811 (database_->sql.prepare << statement.sql, soci::use(parameterValues));
812
813 for (auto& payload : preparedRowSet)
814 {
815 groupedTargets[payload.ownerKey].push_back(std::move(payload.value));
816 }

819 template <typename Owner, typename Options>
820 auto loadIncludedCollections(model::ModelView ownerDescriptor, const Options& queryData,
821 std::vector<Owner>& owners) -> void
822 {
823 if (queryData.includes.empty())
824 {
825 return;
826 }
827
828 for (const auto& includedRelation : queryData.includes)
829 {
830 (void)detail::requireCollectionRelation(ownerDescriptor, includedRelation);
831 }
832
833 if (owners.empty())
834 {
835 return;
836 }
837
838 constexpr auto reflectedFields = reflection::fields<Owner>();
839 auto ownerFields = reflection::fieldPointers(owners.front());
840 auto loadField =
841 [this, &queryData, &owners, ownerDescriptor, &reflectedFields](auto fieldIndex, auto* field)
842 {
843 (void)this;
844 using collection_t = std::decay_t<decltype(*field)>;
845
846 if constexpr (orm::is_relation_collection_v<collection_t>)
847 {
848 const auto relationName = std::string{reflectedFields[fieldIndex].name};
849
850 if (std::ranges::find(queryData.includes, relationName) == queryData.includes.end())
851 {
852 return;
853 }
854
855 const auto* relation = ownerDescriptor.findRelation(relationName);
856 assert(relation != nullptr);
857
858 loadCollectionField<decltype(fieldIndex)::value, Owner, collection_t>(ownerDescriptor, queryData,
859 owners, *relation);
860 }
861 };
862
863 utils::constexpr_for_tuple(ownerFields, loadField);
864 }

866 template <std::size_t FieldIndex, typename Owner, typename Collection, typename Options>
867 auto loadCollectionField(model::ModelView ownerDescriptor, const Options& queryData, std::vector<Owner>& owners,
868 const model::RelationView& relation) -> void
869 {
870 using target_t = orm::relation_target_t<Collection>;
871
872 const auto targetDescriptor =
873 detail::requireCollectionTarget(ownerDescriptor, relation, model::typeId<target_t>());
874
875 const auto ownerPrimaryKey = db::binding::getPrimaryKeyColumns(ownerDescriptor);
876 const auto runtimeLimits = database_->getBackendRuntimeLimits();
877 const auto parameterBudget =
878 runtimeLimits.maxBindParameters.value_or(std::numeric_limits<std::size_t>::max());
879
880 if (parameterBudget < ownerPrimaryKey.size())
881 {
882 throw DatabaseError{DatabaseErrorCode::UnsupportedFeature, database_->backendType, "include collection",
883 "The backend bind-parameter limit is too small for the relation primary key"};
884 }
885
886 const auto batchSize = runtimeLimits.maxBindParameters.has_value() ?
887 std::max<std::size_t>(1, parameterBudget / ownerPrimaryKey.size()) :
888 owners.size();
889 std::map<db::binding::PrimaryKey, std::vector<target_t>, db::binding::PrimaryKeyLess> groupedTargets;
890
891 orm::Query<target_t> targetQuery;
892
893 if (not queryData.shouldJoin)
894 {
895 targetQuery.disableJoining();
896 }
897
898 const auto baseTargetStatement =
899 database_->getCommandGenerator().select(*targetDescriptor, targetQuery.getData());
900
901 assert(baseTargetStatement.parameters.empty());
902
903 for (std::size_t batchStart = 0; batchStart < owners.size(); batchStart += batchSize)
904 {
905 const auto batchEnd = std::min(owners.size(), batchStart + batchSize);
906 std::vector<db::binding::PrimaryKey> ownerKeys;
907 ownerKeys.reserve(batchEnd - batchStart);
908
909 for (auto ownerIndex = batchStart; ownerIndex < batchEnd; ++ownerIndex)
910 {
911 ownerKeys.push_back(db::binding::getPrimaryKey<SchemaType>(owners[ownerIndex]));
912 }
913
914 const auto statement = db::relations::collectionSelectStatement(
915 database_->getBackend().dialect(), ownerDescriptor, relation, baseTargetStatement.sql, ownerKeys,
916 queryData.shouldJoin);
917 if (queryData.shouldJoin)
918 {
919 appendCollectionRows<Owner, target_t, true>(statement, groupedTargets);
920 }
921 else
922 {
923 appendCollectionRows<Owner, target_t, false>(statement, groupedTargets);
924 }
925 }
926
927 for (auto& owner : owners)
928 {
929 const auto ownerKey = db::binding::getPrimaryKey<SchemaType>(owner);
930 auto ownerFields = reflection::fieldPointers(owner);
931 auto* collection = std::get<FieldIndex>(ownerFields);
932 const auto targets = groupedTargets.find(ownerKey);
933
934 if (targets == groupedTargets.end())
935 {
936 collection->setLoaded({});
937 }
938 else
939 {
940 collection->setLoaded(targets->second);
941 }
942 }
943 }
