112 template <typename T>
113 auto select(Query<T>& query) -> std::vector<T>
114 {
115 ensureQuerySupported(query.getData());
116 std::vector<T> result;
117 const auto statement = getCommandGenerator().select(query.getData());
118
119 try
120 {
121 soci::values parameterValues;
122 detail::bindStatementParameters(getBackend().runtime(), parameterValues, statement.parameters);
123
124 auto readRows = [&]<bool JoinedValues>()
125 {
126 soci::rowset<db::binding::BindingPayload<T, JoinedValues>> preparedRowSet =
127 (sql.prepare << statement.sql, soci::use(parameterValues));
128
129 for (auto& payload : preparedRowSet)
130 {
131 result.push_back(std::move(payload.value));
132 }
133 };
134
135 if (query.getData().shouldJoin)
136 {
137 readRows.template operator()<true>();
138 }
139 else
140 {
141 readRows.template operator()<false>();
142 }
143
144 loadIncludedCollections(query.getData(), result);
145 }
146 catch (const db::binding::ConversionError&)
147 {
148 throw DatabaseError{DatabaseErrorCode::Conversion, backendType, "select",
149 "A database result cannot be represented by the requested model"};
150 }
151 catch (const soci::soci_error& error)
152 {
153 throwTranslatedError(error, DatabaseErrorCode::Statement, "select");
154 }
155
156 return result;
157 }

221 template <typename T>
222 auto insert(T object) -> void
223 {
224 const auto& modelInfo = Model<T>::getModelInfo();
225 ensureModelSupported(modelInfo, "insert");
226 requireCapability(getBackendCapabilities().mutations.insert, "insert", "insert is not supported");
227 const auto command = getCommandGenerator().insert(modelInfo);
228
229 const Payload<T> payload{};
230
231 payload.value = std::move(object);
232
233 try
234 {
235 soci::values serializedModel;
236 auto indicator = soci::i_ok;
237 soci::type_conversion<Payload<T>>::to_base(payload, serializedModel, indicator);
238
239 soci::values parameterValues;
240 const auto parameterCount =
241 detail::bindModelParameters(getBackend().runtime(), parameterValues, serializedModel, modelInfo);
242
243 if (parameterCount == 0)
244 {
245 sql << command;
246 }
247 else
248 {
249 sql << command, soci::use(parameterValues);
250 }
251 }
252 catch (const db::binding::ConversionError&)
253 {
254 throw DatabaseError{DatabaseErrorCode::Conversion, backendType, "insert",
255 "A model value cannot be represented by the selected backend"};
256 }
257 catch (const soci::soci_error& error)
258 {
259 throwTranslatedError(error, DatabaseErrorCode::Statement, "insert");

395 template <typename Owner, typename Target>
396 auto link(const Owner& owner, std::string_view relationField, const Target& target) -> std::size_t
397 {
398 const auto& ownerInfo = Model<Owner>::getModelInfo();
399 const auto relation = std::ranges::find_if(ownerInfo.relationsInfo, [&relationField](const auto& candidate)
400 { return candidate.fieldName == relationField; });
401
402 if (relation == ownerInfo.relationsInfo.end() or relation->kind == model::RelationKind::ToOne)
403 {
404 throw std::invalid_argument{"Unknown collection relation: " + std::string{relationField}};
405 }
406
407 if (relation->targetType != std::type_index{typeid(Target)})
408 {
409 throw std::invalid_argument{"Relation target type does not match mapping: " + std::string{relationField}};
410 }
411
412 const auto ownerKey = db::binding::getPrimaryKey(owner);
413 const auto targetKey = db::binding::getPrimaryKey(target);
414
415 if (relation->kind == model::RelationKind::OneToMany)
416 {
417 requireCapability(getBackendCapabilities().relations.oneToMany, "link relation",
418 "one-to-many relations are not supported");
419 }
420 else
421 {
422 requireCapability(getBackendCapabilities().relations.manyToMany, "link relation",
423 "many-to-many relations are not supported");
424 requireCapability(getBackendCapabilities().mutations.atomicInsertIfAbsent, "link relation",
425 "idempotent relation links are not supported");
426 }
427 requireCapability(relation->kind == model::RelationKind::OneToMany ? getBackendCapabilities().mutations.update :
428 getBackendCapabilities().mutations.insert,
429 "link relation", "relation mutations are not supported");
430
431 if (not relationEndpointExists(ownerInfo, ownerKey) or
432 not relationEndpointExists(relation->targetModel(), targetKey))
433 {
434 throw std::invalid_argument{"Cannot link relation endpoints that do not exist"};
435 }
436
437 return executeMutation(
438 db::relations::linkStatement(getBackend().dialect(), ownerInfo, *relation, ownerKey, targetKey),
439 "link relation");
440 }

526 template <typename Owner>
527 auto loadIncludedCollections(const query::QueryData& queryData, std::vector<Owner>& owners) -> void
528 {
529 if (queryData.includes.empty())
530 {
531 return;
532 }
533
534 const auto& ownerInfo = queryData.modelInfo;
535
536 for (const auto& includedRelation : queryData.includes)
537 {
538 const auto* relation = ownerInfo.findRelation(includedRelation);
539
540 if (relation == nullptr or relation->kind == model::RelationKind::ToOne)
541 {
542 throw std::invalid_argument{"Unknown collection relation: " + includedRelation};
543 }
544 }
545
546 if (owners.empty())
547 {
548 return;
549 }
550
551 const auto reflectedFields = rfl::fields<Owner>();
552 auto ownerFields = rfl::to_view(owners.front()).values();
553 auto loadField = [this, &queryData, &owners, &ownerInfo, &reflectedFields](auto fieldIndex, auto* field)
554 {
555 using collection_t = std::decay_t<decltype(*field)>;
556
557 if constexpr (orm::is_relation_collection_v<collection_t>)
558 {
559 const auto relationName = std::string{reflectedFields[fieldIndex].name()};
560
561 if (std::ranges::find(queryData.includes, relationName) == queryData.includes.end())
562 {
563 return;
564 }
565
566 const auto* relation = ownerInfo.findRelation(relationName);
567 assert(relation != nullptr);
568
569 loadCollectionField<decltype(fieldIndex)::value, Owner, collection_t>(queryData, owners, *relation);
570 }
571 };
572
573 utils::constexpr_for_tuple(ownerFields, loadField);
574 }

576 template <std::size_t FieldIndex, typename Owner, typename Collection>
577 auto loadCollectionField(const query::QueryData& queryData, std::vector<Owner>& owners,
578 const model::RelationInfo& relation) -> void
579 {
580 using target_t = orm::relation_target_t<Collection>;
581
582 if (relation.targetType != std::type_index{typeid(target_t)})
583 {
584 throw std::invalid_argument{"Collection wrapper target does not match relation metadata: " +
585 relation.fieldName};
586 }
587
588 const auto ownerPrimaryKey = db::binding::getPrimaryKeyColumns(queryData.modelInfo);
589 const auto runtimeLimits = getBackendRuntimeLimits();
590 const auto parameterBudget = runtimeLimits.maxBindParameters.value_or(std::numeric_limits<std::size_t>::max());
591
592 if (parameterBudget < ownerPrimaryKey.size())
593 {
594 throw DatabaseError{DatabaseErrorCode::UnsupportedFeature, backendType, "include collection",
595 "The backend bind-parameter limit is too small for the relation primary key"};
596 }
597
598 const auto batchSize = runtimeLimits.maxBindParameters.has_value() ?
599 std::max<std::size_t>(1, parameterBudget / ownerPrimaryKey.size()) :
600 owners.size();
601 std::map<db::binding::PrimaryKey, std::vector<target_t>> groupedTargets;
602
603 orm::Query<target_t> targetQuery;
604
605 if (not queryData.shouldJoin)
606 {
607 targetQuery.disableJoining();
608 }
609
610 const auto baseTargetStatement = getCommandGenerator().select(targetQuery.getData());
611
612 assert(baseTargetStatement.parameters.empty());
613
614 for (std::size_t batchStart = 0; batchStart < owners.size(); batchStart += batchSize)
615 {
616 const auto batchEnd = std::min(owners.size(), batchStart + batchSize);
617 std::vector<db::binding::PrimaryKey> ownerKeys;
618 ownerKeys.reserve(batchEnd - batchStart);
619
620 for (auto ownerIndex = batchStart; ownerIndex < batchEnd; ++ownerIndex)
621 {
622 ownerKeys.push_back(db::binding::getPrimaryKey(owners[ownerIndex]));
623 }
624
625 const auto statement =
626 db::relations::collectionSelectStatement(getBackend().dialect(), queryData.modelInfo, relation,
627 baseTargetStatement.sql, ownerKeys, queryData.shouldJoin);
628 if (queryData.shouldJoin)
629 {
630 appendCollectionRows<Owner, target_t, true>(statement, groupedTargets);
631 }
632 else
633 {
634 appendCollectionRows<Owner, target_t, false>(statement, groupedTargets);
635 }
636 }
637
638 for (auto& owner : owners)
639 {
640 const auto ownerKey = db::binding::getPrimaryKey(owner);
641 auto ownerFields = rfl::to_view(owner).values();
642 auto* collection = std::get<FieldIndex>(ownerFields);
643 const auto targets = groupedTargets.find(ownerKey);
644
645 if (targets == groupedTargets.end())
646 {
647 collection->setLoaded({});
648 }
649 else
650 {
651 collection->setLoaded(targets->second);
652 }
653 }
654 }
