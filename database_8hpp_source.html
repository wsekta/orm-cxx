41inline auto bindNullStatementParameter(soci::values& values, const db::StatementParameter& parameter) -> void
42{
43 switch (parameter.nullType)
44 {
45 case model::ColumnType::Bool:
46 case model::ColumnType::Char:
47 case model::ColumnType::UnsignedChar:
48 case model::ColumnType::Short:
49 case model::ColumnType::UnsignedShort:
50 case model::ColumnType::Int:
51 bindTypedNullStatementParameter(values, parameter.name, int{});
52 return;
53 case model::ColumnType::UnsignedInt:
54 case model::ColumnType::UnsignedLongLong:
55 bindTypedNullStatementParameter(values, parameter.name, static_cast<unsigned long long>(0));
56 return;
57 case model::ColumnType::LongLong:
58 bindTypedNullStatementParameter(values, parameter.name, static_cast<long long>(0));
59 return;
60 case model::ColumnType::Float:
61 case model::ColumnType::Double:
62 bindTypedNullStatementParameter(values, parameter.name, 0.0);
63 return;
64 case model::ColumnType::String:
65 bindTypedNullStatementParameter(values, parameter.name, std::string{});
66 return;
67 case model::ColumnType::Uuid:
68 case model::ColumnType::Unknown:
69 case model::ColumnType::OneToOne:
70 throw std::invalid_argument{"Cannot bind NULL parameter with unsupported column type"};
71 }
72
73 throw std::invalid_argument{"Cannot bind NULL parameter with unsupported column type"};
74}

76inline auto bindStatementParameter(soci::values& values, const db::StatementParameter& parameter) -> void
77{
78 if (parameter.value.has_value())
79 {
80 std::visit([&values, &parameter](const auto& value) { values.set(parameter.name, value); },
81 parameter.value->get());
82 return;
83 }
84
85 bindNullStatementParameter(values, parameter);
86}

88inline auto bindStatementParameters(soci::values& values, const std::vector<db::StatementParameter>& parameters) -> void
89{
90 for (const auto& parameter : parameters)
91 {
92 bindStatementParameter(values, parameter);

340 template <typename Owner, typename Target>
341 auto link(const Owner& owner, std::string_view relationField, const Target& target) -> std::size_t
342 {
343 const auto& ownerInfo = Model<Owner>::getModelInfo();
344 const auto relation = std::ranges::find_if(ownerInfo.relationsInfo, [&relationField](const auto& candidate)
345 { return candidate.fieldName == relationField; });
346
347 if (relation == ownerInfo.relationsInfo.end() or relation->kind == model::RelationKind::ToOne)
348 {
349 throw std::invalid_argument{"Unknown collection relation: " + std::string{relationField}};
350 }
351
352 if (relation->targetType != std::type_index{typeid(Target)})
353 {
354 throw std::invalid_argument{"Relation target type does not match mapping: " + std::string{relationField}};
355 }
356
357 const auto ownerKey = db::binding::getPrimaryKey(owner);
358 const auto targetKey = db::binding::getPrimaryKey(target);
359
360 if (not relationEndpointExists(ownerInfo, ownerKey) or
361 not relationEndpointExists(relation->targetModel(), targetKey))
362 {
363 throw std::invalid_argument{"Cannot link relation endpoints that do not exist"};
364 }
365
366 return executeMutation(db::relations::linkStatement(ownerInfo, *relation, ownerKey, targetKey));
367 }

432 template <typename Owner>
433 auto loadIncludedCollections(const query::QueryData& queryData, std::vector<Owner>& owners) -> void
434 {
435 if (queryData.includes.empty())
436 {
437 return;
438 }
439
440 const auto& ownerInfo = queryData.modelInfo;
441
442 for (const auto& includedRelation : queryData.includes)
443 {
444 const auto* relation = ownerInfo.findRelation(includedRelation);
445
446 if (relation == nullptr or relation->kind == model::RelationKind::ToOne)
447 {
448 throw std::invalid_argument{"Unknown collection relation: " + includedRelation};
449 }
450 }
451
452 if (owners.empty())
453 {
454 return;
455 }
456
457 const auto reflectedFields = rfl::fields<Owner>();
458 auto ownerFields = rfl::to_view(owners.front()).values();
459 auto loadField = [this, &queryData, &owners, &ownerInfo, &reflectedFields](auto fieldIndex, auto* field)
460 {
461 using collection_t = std::decay_t<decltype(*field)>;
462
463 if constexpr (orm::is_relation_collection_v<collection_t>)
464 {
465 const auto relationName = std::string{reflectedFields[fieldIndex].name()};
466
467 if (std::ranges::find(queryData.includes, relationName) == queryData.includes.end())
468 {
469 return;
470 }
471
472 const auto* relation = ownerInfo.findRelation(relationName);
473 assert(relation != nullptr);
474
475 loadCollectionField<decltype(fieldIndex)::value, Owner, collection_t>(queryData, owners, *relation);
476 }
477 };
478
479 utils::constexpr_for_tuple(ownerFields, loadField);
480 }

482 template <std::size_t FieldIndex, typename Owner, typename Collection>
483 auto loadCollectionField(const query::QueryData& queryData, std::vector<Owner>& owners,
484 const model::RelationInfo& relation) -> void
485 {
486 using target_t = orm::relation_target_t<Collection>;
487
488 if (relation.targetType != std::type_index{typeid(target_t)})
489 {
490 throw std::invalid_argument{"Collection wrapper target does not match relation metadata: " +
491 relation.fieldName};
492 }
493
494 const auto ownerPrimaryKey = db::binding::getPrimaryKeyColumns(queryData.modelInfo);
495 constexpr std::size_t parameterBudget = 900;
496 const auto batchSize = std::max<std::size_t>(1, parameterBudget / ownerPrimaryKey.size());
497 std::map<db::binding::PrimaryKey, std::vector<target_t>> groupedTargets;
498
499 orm::Query<target_t> targetQuery;
500
501 if (not queryData.shouldJoin)
502 {
503 targetQuery.disableJoining();
504 }
505
506 const auto baseTargetStatement =
507 commandGeneratorFactory.getCommandGenerator(backendType).select(targetQuery.getData());
508
509 assert(baseTargetStatement.parameters.empty());
510
511 for (std::size_t batchStart = 0; batchStart < owners.size(); batchStart += batchSize)
512 {
513 const auto batchEnd = std::min(owners.size(), batchStart + batchSize);
514 std::vector<db::binding::PrimaryKey> ownerKeys;
515 ownerKeys.reserve(batchEnd - batchStart);
516
517 for (auto ownerIndex = batchStart; ownerIndex < batchEnd; ++ownerIndex)
518 {
519 ownerKeys.push_back(db::binding::getPrimaryKey(owners[ownerIndex]));
520 }
521
522 const auto statement = db::relations::collectionSelectStatement(
523 queryData.modelInfo, relation, baseTargetStatement.sql, ownerKeys, queryData.shouldJoin);
524 if (queryData.shouldJoin)
525 {
526 appendCollectionRows<Owner, target_t, true>(statement, groupedTargets);
527 }
528 else
529 {
530 appendCollectionRows<Owner, target_t, false>(statement, groupedTargets);
531 }
532 }
533
534 for (auto& owner : owners)
535 {
536 const auto ownerKey = db::binding::getPrimaryKey(owner);
537 auto ownerFields = rfl::to_view(owner).values();
538 auto* collection = std::get<FieldIndex>(ownerFields);
539 const auto targets = groupedTargets.find(ownerKey);
540
541 if (targets == groupedTargets.end())
542 {
543 collection->setLoaded({});
544 }
545 else
546 {
547 collection->setLoaded(targets->second);
548 }
549 }
550 }
