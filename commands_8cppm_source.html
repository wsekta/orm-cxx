37/** One bound insert column: its logical parameter name and the physical column it maps to. */
38struct InsertColumnBinding
39{
40 std::string logicalName;
41 const model::ColumnView* column;
42};
43
44/**
45 * Internal mapping from a model to the columns a plain or batch insert binds, in statement order.
46 *
47 * Auto-increment columns are omitted; a to-one column contributes one logical name per primary-key
48 * column of its target ("field_pk") and binds the target column's type. Legacy insert, insertMany
49 * and the parameter binders all use this single derivation. This is an implementation seam, not
50 * part of the documented backend-extension surface.
51 */
52[[nodiscard]] inline auto insertColumnBindings(model::ModelView model) -> std::vector<InsertColumnBinding>
53{
54 std::vector<InsertColumnBinding> bindings;
55
56 for (const auto& column : model->columns)
57 {
58 if (column.kind == model::FieldKind::ToOne)
59 {
60 const auto target = model.resolveTarget(column);
61 if (target == nullptr)
62 {
63 throw std::invalid_argument{"To-one column has no target model in the schema"};
64 }
65
66 for (const auto& relatedColumn : target->columns)
67 {
68 if (relatedColumn.isPrimaryKey)
69 {
70 bindings.push_back({std::format("{}_{}", column.name, relatedColumn.name), &relatedColumn});
71 }
72 }
73 }
74 else if (not column.isAutoIncrement)
75 {
76 bindings.push_back({std::string{column.name}, &column});
77 }
78 }
79
80 return bindings;
81}
