439inline auto isWithinValueLimits(const BoundValue& bound, const BackendValueLimits& limits,
440 const BackendRuntimeLimits& runtime = {}) noexcept -> bool
441{
442 if (bound.isNull())
443 return true;
444 if (!query::QueryValue::isCompatibleStorage(bound.logicalType, *bound.value))
445 return false;
446 const auto within = [](const auto& value, const auto& minimum, const auto& maximum)
447 { return (!minimum || value >= *minimum) && (!maximum || value <= *maximum); };
448 return std::visit(
449 [&](const auto& value)
450 {
451 using T = std::decay_t<decltype(value)>;
452 if constexpr (std::is_same_v<T, Date>)
453 return within(value, limits.minDate, limits.maxDate);
454 else if constexpr (std::is_same_v<T, LocalDateTime>)
455 return within(value, limits.minLocalDateTime, limits.maxLocalDateTime);
456 else if constexpr (std::is_same_v<T, UtcTimestamp>)
457 return within(value, limits.minUtcTimestamp, limits.maxUtcTimestamp) &&
458 within(value, runtime.minUtcTimestamp, runtime.maxUtcTimestamp);
459 else if constexpr (std::is_same_v<T, unsigned long long>)
460 return !limits.maxUnsignedLongLong || value <= *limits.maxUnsignedLongLong;
461 else
462 return true;
463 },
464 *bound.value);
465}
