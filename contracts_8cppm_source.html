228 struct StatementParameter
229 {
230 std::string name;
231 std::optional<query::QueryValue> value;
232 std::optional<model::ColumnType> nullType;
233
234 [[nodiscard]] auto getLogicalType() const -> model::ColumnType
235 {
236 if (value.has_value())
237 {
238 return value->getLogicalType();
239 }
240 if (nullType.has_value())
241 {
242 return nullType.value();
243 }
244 throw std::invalid_argument{"A NULL statement parameter requires an explicit logical type"};
245 }
246
247 [[nodiscard]] auto getBoundValue() const -> BoundValue
248 {
249 if (value.has_value())
250 {
251 return BoundValue{.logicalType = value->getLogicalType(), .value = value->get()};
252 }
253
254 return BoundValue{.logicalType = getLogicalType(), .value = std::nullopt};
255 }
256 };
