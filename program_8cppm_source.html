167[[nodiscard]] constexpr auto decimal(std::size_t value) -> std::string
168{
169 std::string result;
170 do
171 {
172 result.push_back(static_cast<char>('0' + value % 10));
173 value /= 10;
174 } while (value != 0);
175 std::ranges::reverse(result);
176 return result;

189[[nodiscard]] constexpr auto quoteStandardIdentifier(std::string_view identifier,
190 CompiledSqlFlavor flavor) -> std::string
191{
192 if (identifier.empty())
193 throw std::invalid_argument{"SQL identifier cannot be empty"};
194 if (identifier.find('\0') != std::string_view::npos)
195 throw std::invalid_argument{"SQL identifiers must not contain an embedded NUL byte"};
196 if (flavor == CompiledSqlFlavor::PostgreSQL && identifier.size() > 63)
197 throw std::invalid_argument{"PostgreSQL identifiers must not exceed 63 bytes"};
198 if (flavor == CompiledSqlFlavor::MySQLFamily &&
199 std::ranges::count_if(identifier, [](unsigned char c) { return (c & 0xc0) != 0x80; }) > 64)
200 throw std::invalid_argument{"MySQL-family identifiers must not exceed 64 characters"};
201 const auto quote = flavor == CompiledSqlFlavor::MySQLFamily ? '`' : '"';
202 std::string result(1, quote);
203 for (const auto character : identifier)
204 {
205 if (character == quote)
206 result += quote;
207 result += character;
208 }
209 result += quote;
210 return result;
211}
