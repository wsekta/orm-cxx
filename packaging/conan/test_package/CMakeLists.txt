cmake_minimum_required(VERSION 3.31)
project(orm-cxx-test-package LANGUAGES CXX)

find_package(orm-cxx CONFIG REQUIRED)

add_executable(test_package test_package.cpp)
target_compile_features(test_package PRIVATE cxx_std_20)
target_link_libraries(test_package PRIVATE orm-cxx::orm-cxx)
foreach(backend SQLITE POSTGRESQL MYSQL MARIADB)
    string(TOLOWER "${backend}" backend_lower)
    set(orm_cxx_expected_${backend_lower} std::nullopt)
    if(DEFINED ORM_CXX_EXPECT_${backend})
        if(ORM_CXX_EXPECT_${backend})
            set(orm_cxx_expected_${backend_lower} true)
        else()
            set(orm_cxx_expected_${backend_lower} false)
        endif()
    endif()
endforeach()

enable_testing()
include(BackendCompileContracts.cmake)
orm_cxx_backend_compile_contracts(test_package)
configure_file(BackendExpectations.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/BackendExpectations.hpp" @ONLY)
target_include_directories(test_package PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

add_executable(test_reflection test_reflection.cpp)
target_compile_features(test_reflection PRIVATE cxx_std_20)
target_link_libraries(test_reflection PRIVATE orm-cxx::reflection)

foreach(consumer IN ITEMS test_package test_reflection)
    if(MSVC)
        target_compile_options(${consumer} PRIVATE /W4 /WX /wd4996)
    else()
        target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
