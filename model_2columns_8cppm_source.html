64 for (std::size_t right = left + 1; right < names.size(); ++right)
65 if (names[left] == names[right] && backends[left] == backends[right])
66 return false;

321template <typename Model>
322consteval auto normalizedDefaultTableName()
323{
324 constexpr auto source = reflection::typeNameStorage<Model>.view();
325 constexpr auto prefixSize = source.starts_with("class ") ? std::string_view{"class "}.size() :
326 source.starts_with("struct ") ? std::string_view{"struct "}.size() :
327 0U;
328 constexpr auto resultSize = [source]
329 {
330 std::size_t result{};
331 for (std::size_t index = prefixSize; index < source.size(); ++index)
332 {
333 ++result;
334 if (source[index] == ':' && index + 1 < source.size() && source[index + 1] == ':')
335 {
336 ++index;
337 }
338 }
339 return result;
340 }();
341
342 std::array<char, resultSize + 1> result{};
343 std::size_t output{};
344 for (std::size_t index = prefixSize; index < source.size(); ++index)
345 {
346 if (source[index] == ':' && index + 1 < source.size() && source[index + 1] == ':')
347 {
348 result[output++] = '_';
349 ++index;
350 }
351 else
352 {
353 result[output++] = source[index];
354 }
355 }
356 return result;
357}

466template <typename Model, typename SchemaType, std::size_t Index>
467consteval auto makeColumn() -> ColumnView
468{
469 using field_t = reflection::field_type_t<Model, Index>;
470 using value_t = std::remove_cv_t<static_optional_value_t<std::remove_cvref_t<field_t>>>;
471 constexpr auto fieldName = reflectedFieldName<Model, Index>();
472 static_assert(not is_optional_relation_collection_v<field_t>,
473 "ORM_MODEL_OPTIONAL_COLLECTION: relation collections cannot be optional");
474
475 if constexpr (StaticSchemaTraits<SchemaType>::template contains<value_t>)
476 {
477 static_assert(hasPrimaryKey<value_t>(),
478 "ORM_RELATION_KEYLESS: every to-one target must define a valid primary key");
479 return ColumnView{
480 .fieldIndex = Index,
481 .fieldName = fieldName,
482 .name = mappedColumnName<Model>(fieldName),
483 .type = std::nullopt,
484 .kind = FieldKind::ToOne,
485 .targetModelIndex = StaticSchemaTraits<SchemaType>::template indexOf<value_t>(),
486 .isPrimaryKey = isPrimaryKey<Model>(fieldName),
487 .isAutoIncrement = isAutoIncrement<Model>(fieldName),
488 .isNotNull = not isNullable<field_t>,
489 .maxStringLength = std::nullopt,
490 };
491 }
492 else
493 {
494 return ColumnView{
495 .fieldIndex = Index,
496 .fieldName = fieldName,
497 .name = mappedColumnName<Model>(fieldName),
498 .type = logicalType<field_t>(),
499 .kind = FieldKind::Scalar,
500 .targetModelIndex = noTargetModel,
501 .isPrimaryKey = isPrimaryKey<Model>(fieldName),
502 .isAutoIncrement = isAutoIncrement<Model>(fieldName),
503 .isNotNull = not isNullable<field_t>,
504 .maxStringLength = mappedStringLength<Model>(fieldName),
505 .mysqlUuidStorage = mappedUuidStorage<Model>(fieldName, db::BackendType::Mysql),
506 .sqliteUuidStorage = mappedUuidStorage<Model>(fieldName, db::BackendType::Sqlite),
507 };
508 }

511template <typename Model, std::size_t OutputIndex, std::size_t... FieldIndices>
512consteval auto columnFieldIndex(std::index_sequence<FieldIndices...>) -> std::size_t
513{
514 constexpr std::array fieldIndices{FieldIndices...};
515 constexpr std::array included{

517 not is_optional_relation_collection_v<reflection::field_type_t<Model, FieldIndices>>)...};
518 std::size_t output{};
519 std::size_t result = reflection::fieldCount<Model>;
520 for (std::size_t index = 0; index < included.size(); ++index)
521 {
522 if (included[index])
523 {

525 {
526 result = fieldIndices[index];

528 }

596{
