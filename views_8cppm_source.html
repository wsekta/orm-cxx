289constexpr auto ModelView::resolveJunction(const RelationView& relation) const noexcept -> JunctionView
290{
291 if (relation.junction.isConfigured())
292 {
293 return relation.junction;
294 }
295 if (relation.kind != RelationKind::ManyToMany || relation.mappedBy.empty())
296 {
297 return {};
298 }
299
300 const auto target = resolveTarget(relation);
301 const auto* owningRelation = target ? target.findRelationField(relation.mappedBy) : nullptr;
302 if (owningRelation == nullptr || owningRelation->kind != RelationKind::ManyToMany ||
303 not owningRelation->junction.isConfigured())
304 {
305 return {};
306 }
307
308 return JunctionView{
309 .tableName = owningRelation->junction.tableName,
310 .ownerColumns = owningRelation->junction.targetColumns,
311 .targetColumns = owningRelation->junction.ownerColumns,
312 .owningSide = false,
313 };
314}
